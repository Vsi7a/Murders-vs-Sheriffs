-- ЛОГИКА C-FLY
local cflyConnection
local originalGravity = workspace.Gravity
local storedCFlyCFrame = nil

local function toggleCFly(state)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChildOfClass("Humanoid") then return end
    
    local rootPart = char.HumanoidRootPart
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    
    FlyUpBtn.Visible = state
    FlyDownBtn.Visible = state
    
    if state then
        workspace.Gravity = 0
        humanoid.PlatformStand = true
        storedCFlyCFrame = rootPart.CFrame
        
        cflyConnection = RunService.RenderStepped:Connect(function(dt)
            if not getgenv().TWKS_Config.CFly then return end
            
            local cam = CurrentCamera
            local moveDir = Vector3.new(0, 0, 0)
            local moveVector = humanoid.MoveDirection
            
            if moveVector.Magnitude > 0 then
                local camLook = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z).Unit
                local camRight = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z).Unit
                
                moveDir = (camLook * moveVector.Z) + (camRight * moveVector.X)
            end
            
            if isFlyingUp then moveDir = moveDir + Vector3.new(0, 1, 0) end
            if isFlyingDown then moveDir = moveDir - Vector3.new(0, 1, 0) end
            
            if moveDir.Magnitude > 0 then
                storedCFlyCFrame = storedCFlyCFrame + (moveDir.Unit * getgenv().TWKS_Config.CFlySpeed * dt)
            end
            
            rootPart.CFrame = storedCFlyCFrame
            rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end)
    else
        if cflyConnection then
            cflyConnection:Disconnect()
            cflyConnection = nil
        end
        workspace.Gravity = originalGravity
        if humanoid then humanoid.PlatformStand = false end
        storedCFlyCFrame = nil
        isFlyingUp = false
        isFlyingDown = false
    end
end

local function performClick()
    local viewport = CurrentCamera.ViewportSize
    local targetX = (viewport.X / 2) + getgenv().TWKS_Config.ClickOffsetX
    local targetY = (viewport.Y / 2) + getgenv().TWKS_Config.ClickOffsetY
    
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, true, game, 0)
        task.wait(0.05)
        VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, false, game, 0)
    end)
end

    if (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1) and not activeTouchInput then

    end

end)

-- ПОИСК ЦЕЛЕЙ И ПРОВЕРКА КОМАНД СЕРВЕРА
local function getTargetCharacters()
    local characters = {}
    pcall(function()
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj ~= LocalPlayer.Character then
                local humanoid = obj:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    table.insert(characters, obj)
                end
            end
        end
    end)
    return characters
end

local function isEnemy(playerModel)
    if playerModel == LocalPlayer.Character then return false end
    
    local targetPlayer = Players:GetPlayerFromCharacter(playerModel)
    if targetPlayer then
        local myTeam = cachedTeams[LocalPlayer.UserId]
        local targetTeam = cachedTeams[targetPlayer.UserId]
        
        if myTeam and targetTeam then
            if myTeam == targetTeam then return false else return true end
        end
    end
    
    return true
end

local function isVisible(targetPart)
    if not getgenv().TWKS_Config.WallCheck then return true end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("Head") then return false end
    
    local origin = myChar.Head.Position
    local direction = targetPart.Position - origin
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {myChar, targetPart.Parent}
    
    local success, result = pcall(function()
        return workspace:Raycast(origin, direction, raycastParams)
    end)
    
    return success and result == nil
end

local function getClosestEnemy()
    local target = nil
    local shortestDistance = math.huge
    
    for _, char in ipairs(getTargetCharacters()) do
        if isEnemy(char) then
            local targetPart = char:FindFirstChild("HumanoidRootPart") 
                or char:FindFirstChild("Head") 
                or char:FindFirstChildOfClass("BasePart")
                
            if targetPart and isVisible(targetPart) then
                local success, screenPoint, onScreen = pcall(function()
                    return CurrentCamera:WorldToViewportPoint(targetPart.Position)
                end)
                
                if success and onScreen then
                    local mouse = LocalPlayer:GetMouse()
                    local dist = (Vector2.new(mouse.X, mouse.Y) - Vector2.new(screenPoint.X, screenPoint.Y)).Magnitude
                    if dist < shortestDistance then
                        shortestDistance = dist
                        target = targetPart
                    end
                end
            end
        end
    end
    return target
end

-- ESP РИСУНКИ И ОЧИСТКА
local espDrawings = {}

local function clearESP()
    local function clearESP()
    for _, drawings in pairs(espDrawings) do
        pcall(function()
            if drawings.Box then drawings.Box:Remove() end
            if drawings.Line then drawings.Line:Remove() end
            if drawings.Name then drawings.Name:Remove() end
        end)
    end
    espDrawings = {}
end

local function removeCharESP(char)
    if espDrawings[char] then
        pcall(function()
            if espDrawings[char].Box then espDrawings[char].Box:Remove() end
            if espDrawings[char].Line then espDrawings[char].Line:Remove() end
            if espDrawings[char].Name then espDrawings[char].Name:Remove() end
        end)
        espDrawings[char] = nil
    end
end

-- ОСНОВНОЙ ЦИКЛ РАБОТЫ
RunService.RenderStepped:Connect(function()
    local cfg = getgenv().TWKS_Config
    local targetPart = getClosestEnemy()
    
    if cfg.CFly and not cflyConnection then
        toggleCFly(true)
    elseif not cfg.CFly and cflyConnection then
        toggleCFly(false)
    end
    
    if cfg.Aimbot and targetPart then
        pcall(function()
            local currentPos = CurrentCamera.CFrame.Position
            local targetCFrame = CFrame.lookAt(currentPos, targetPart.Position)
            CurrentCamera.CFrame = CurrentCamera.CFrame:Lerp(targetCFrame, cfg.Sensitivity)
        end)
    end
    
    -- ПРОВЕРКА НАВОДКИ ДЛЯ ТРИГГЕРА (Срабатывает только если прицел наведен на врага)
    if cfg.Triggerbot and targetPart then
        local success, screenPoint, onScreen = pcall(function()
            return CurrentCamera:WorldToViewportPoint(targetPart.Position)
        end)
        
        if success and onScreen then
            local viewport = CurrentCamera.ViewportSize
            local screenCenter = Vector2.new(viewport.X / 2, viewport.Y / 2)
            local targetScreenPos = Vector2.new(screenPoint.X, screenPoint.Y)
            
            -- Допустимый радиус наводки от центра экрана (в пикселях). 
            -- Увеличен до 35 пикселей, чтобы триггер стабильно срабатывал при наведении на хитбокс
            local crosshairDistance = (screenCenter - targetScreenPos).Magnitude
            
            if crosshairDistance <= 35 then
                performClick()
            end
        end
    end
    
    local targets = getTargetCharacters()
    local activeChars = {}
    for _, c in ipairs(targets) do activeChars[c] = true end
    
    for char, _ in pairs(espDrawings) do
        if not activeChars[char] or not char.Parent then
            removeCharESP(char)
        end
    end
    
    for _, char in ipairs(targets) do
        pcall(function()
            if cfg.Chams then
                local highlight = char:FindFirstChild("TWKS_Chams")
                if not highlight then
                    highlight = Instance.new("Highlight")
                    highlight.Name = "TWKS_Chams"
                    highlight.Adornee = char
                    highlight.FillColor = cfg.ChamsColor
                    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                    highlight.FillTransparency = 0.5
                    highlight.Parent = char
                end
            else
                local highlight = char:FindFirstChild("TWKS_Chams")
                if highlight then highlight:Destroy() end
            end
        end)
    end
    
    for _, char in ipairs(targets) do
        pcall(function()
            local part = char:FindFirstChild(cfg.HitboxPart) or char:FindFirstChildOfClass("BasePart")
            if part then
                if not originalSizes[part] then
                    originalSizes[part] = part.Size
                end
                
                if cfg.HitboxEnabled and cfg.HitboxSize > 1 then
                    part.Size = originalSizes[part] * cfg.HitboxSize
                    part.Transparency = 0.7
                    part.CanCollide = false
                else
                    part.Size = originalSizes[part]
                    part.Transparency = 0
                end
            end
        end)
    end
    
    if not cfg.ESP_Enabled then
        clearESP()
    else
        for _, char in ipairs(targets) do
            pcall(function()
                local primaryPart = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildOfClass("BasePart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if primaryPart and humanoid and humanoid.Health > 0 then
                    if not espDrawings[char] then
                        espDrawings[char] = {
                            Box = Drawing.new("Square"),
                            Line = Drawing.new("Line"),
                            Name = Drawing.new("Text")
                        }
                        espDrawings[char].Box.Thickness = 1
                        espDrawings[char].Box.Color = Color3.fromRGB(255, 255, 255)
                        espDrawings[char].Box.Filled = false
                        
                        espDrawings[char].Line.Thickness = 1
                        espDrawings[char].Line.Color = Color3.fromRGB(255, 255, 255)
                        
                        espDrawings[char].Name.Size = 14
                        espDrawings[char].Name.Center = true
                        espDrawings[char].Name.Outline = true
                        espDrawings[char].Name.Color = Color3.fromRGB(255, 255, 255)
                    end
                    
                    local d = espDrawings[char]
                    local pos, onScreen = CurrentCamera:WorldToViewportPoint(primaryPart.Position + Vector3.new(cfg.OffsetX, cfg.OffsetY, 0))
                    
                    if onScreen and pos.Z > 0 then
                        if cfg.Box2D then
                            d.Box.Visible = true
                            d.Box.Size = Vector2.new(40, 60)
                            d.Box.Position = Vector2.new(pos.X - 20, pos.Y - 30)
                        else
                            d.Box.Visible = false
                        end
                        
                        if cfg.Lines then
                            d.Line.Visible = true
                            d.Line.From = Vector2.new(CurrentCamera.ViewportSize.X / 2, CurrentCamera.ViewportSize.Y)
                            d.Line.To = Vector2.new(pos.X, pos.Y)
                        else
                            d.Line.Visible = false
                        end
                        
                        if cfg.Names then
                            d.Name.Visible = true
                            d.Name.Text = char.Name
                            d.Name.Position = Vector2.new(pos.X, pos.Y - 45)
                        else
                            d.Name.Visible = false
                        end
                    else
                        d.Box.Visible = false
                        d.Line.Visible = false
                        d.Name.Visible = false
                    end
                else
                    removeCharESP(char)
                end
            end)
        end
    end
end)
