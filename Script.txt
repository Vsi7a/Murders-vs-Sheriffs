local function hexToColor3(hex)
    hex = hex:gsub("#", "")
    if #hex ~= 6 then return Color3.new(1, 1, 1) end
    local r = tonumber("0x" .. hex:sub(1, 2)) or 255
    local g = tonumber("0x" .. hex:sub(3, 4)) or 255
    local b = tonumber("0x" .. hex:sub(5, 6)) or 255
    return Color3.fromRGB(r, g, b)
end

local configFileName = "TWKS_NewConfig_Sliders.json"
local activeFloatingButtons = {}

local function saveConfig()
    pcall(function()
        for k, btn in pairs(activeFloatingButtons) do
            if btn and btn.AbsolutePosition then
                getgenv().TWKS_Config.ButtonPositions[k] = {btn.Position.X.Scale, btn.Position.X.Offset, btn.Position.Y.Scale, btn.Position.Y.Offset}
            end
        end
        local data = HttpService:JSONEncode(getgenv().TWKS_Config)
        if writefile then writefile(configFileName, data) end
    end)
end

local function loadConfig()
    pcall(function()
        if isfile and isfile(configFileName) then
            local content = readfile(configFileName)
            local data = HttpService:JSONDecode(content)
            if data and type(data) == "table" then
                for k, v in pairs(data) do
                    getgenv().TWKS_Config[k] = v
                end
            end
        end
    end)
end

-- Настоящий интерактивный ползунок (Slider)
local function addSlider(parent, text, configKey, minVal, maxVal)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 0, 42)
    frame.BackgroundTransparency = 1
    frame.Parent = parent
    
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, 0, 0, 18)
    lbl.BackgroundTransparency = 1
    lbl.Text = "   " .. text .. ": " .. tostring(getgenv().TWKS_Config[configKey])
    lbl.TextColor3 = Color3.fromRGB(200, 200, 200)
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = frame
    
    local sliderBar = Instance.new("Frame")
    sliderBar.Size = UDim2.new(1, -20, 0, 8)
    sliderBar.Position = UDim2.new(0, 10, 0, 24)
    sliderBar.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    sliderBar.BorderSizePixel = 0
    sliderBar.Parent = frame
    Instance.new("UICorner", sliderBar).CornerRadius = UDim.new(1, 0)
    
    local fillBar = Instance.new("Frame")
    fillBar.Size = UDim2.new((getgenv().TWKS_Config[configKey] - minVal) / (maxVal - minVal), 0, 1, 0)
    fillBar.BackgroundColor3 = Color3.fromRGB(50, 150, 255)
    fillBar.BorderSizePixel = 0
    fillBar.Parent = sliderBar
    Instance.new("UICorner", fillBar).CornerRadius = UDim.new(1, 0)
    
    local knob = Instance.new("TextButton")
    knob.Size = UDim2.new(0, 14, 0, 18)
    knob.AnchorPoint = Vector2.new(0.5, 0.5)
    knob.Position = UDim2.new(fillBar.Size.X.Scale, 0, 0.5, 0)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.Text = ""
    knob.Parent = sliderBar
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
    
    local dragging = false
    
    local function updateValue(input)
        local pos = math.clamp((input.Position.X - sliderBar.AbsolutePosition.X) / sliderBar.AbsoluteSize.X, 0, 1)
        local val = math.floor(minVal + ((maxVal - minVal) * pos))
        fillBar.Size = UDim2.new(pos, 0, 1, 0)
        knob.Position = UDim2.new(pos, 0, 0.5, 0)
        getgenv().TWKS_Config[configKey] = val
        lbl.Text = "   " .. text .. ": " .. tostring(val)
        saveConfig()
    end
    
    knob.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
        end
    end)
    
    sliderBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateValue(input)
        end
    end)
    
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateValue(input)
        end
    end)
end

local function addHexColorInput(parent, labelText, configKey)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 0, 32)
    frame.BackgroundTransparency = 1
    frame.Parent = parent
    
    local lbl = Instance.new("TextLabel")

    lbl.BackgroundTransparency = 1
    lbl.Text = "   " .. labelText
    lbl.TextColor3 = Color3.fromRGB(200, 200, 200)
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = frame
    
    local box = Instance.new("TextBox")
    box.Size = UDim2.new(0, 80, 0, 24)
    box.Position = UDim2.new(0.6, 0, 0, 4)
    box.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    box.Text = tostring(getgenv().TWKS_Config[configKey])
    box.TextColor3 = Color3.fromRGB(255, 255, 255)
    box.Font = Enum.Font.GothamBold
    box.TextSize = 11
    box.Parent = frame
    Instance.new("UICorner", box).CornerRadius = UDim.new(0, 4)
    
    box.FocusLost:Connect(function()
        local val = box.Text
        if not val:match("^#") then val = "#" .. val end
        if #val == 7 then
            getgenv().TWKS_Config[configKey] = val
            box.Text = val
            saveConfig()
        else
            box.Text = getgenv().TWKS_Config[configKey]
        end
    end)
end

-- СОЗДАНИЕ ПЛАВАЮЩИХ КНОПОК
local function createFloatingButton(name, defaultPosY)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 50, 0, 50)
    local savedPos = getgenv().TWKS_Config.ButtonPositions[name]
    if savedPos then
        btn.Position = UDim2.new(savedPos[1], savedPos[2], savedPos[3], savedPos[4])
    else
        btn.Position = UDim2.new(0.03, 0, 0.15, defaultPosY)
    end
    btn.BackgroundColor3 = Color3.fromRGB(100, 50, 200)
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 9
    btn.Visible = false
    btn.Active = true
    btn.Draggable = true
    btn.Parent = ScreenGui
    Instance.new("UICorner", btn).CornerRadius = UDim.new(1, 0)
    
    btn.MouseButton1Click:Connect(function()
        if getgenv().TWKS_Config[name] ~= nil then
            getgenv().TWKS_Config[name] = not getgenv().TWKS_Config[name]
        end
    end)
    
    activeFloatingButtons[name] = btn
end

CenterClickBtn.Draggable = true
CenterClickBtn.Parent = ScreenGui
Instance.new("UICorner", CenterClickBtn).CornerRadius = UDim.new(1, 0)

local function performClick()
    local viewport = CurrentCamera.ViewportSize
    local targetX = (viewport.X / 2) + getgenv().TWKS_Config.ClickOffsetX
    local targetY = (viewport.Y / 2) + getgenv().TWKS_Config.ClickOffsetY
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, true, game, 0)
        task.wait(0.05)
        VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, false, game, 0)
    end)
end

CenterClickBtn.MouseButton1Click:Connect(performClick)

-- ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ И ЛОГИКА
local function getTargetCharacters()
    local characters = {}
    pcall(function()
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj ~= LocalPlayer.Character then
                local humanoid = obj:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    table.insert(characters, obj)
                end
            end
        end
    end)
    return characters
end

local function isEnemy(playerModel)
    if playerModel == LocalPlayer.Character then return false end
    local targetPlayer = Players:GetPlayerFromCharacter(playerModel)
    if targetPlayer then
        local myTeam = cachedTeams[LocalPlayer.UserId]
        local targetTeam = cachedTeams[targetPlayer.UserId]
        if myTeam and targetTeam then
            if myTeam == targetTeam then return false else return true end
        end
    end
    return true
end

local function isVisible(targetPart, wallCheckEnabled)
    if not wallCheckEnabled then return true end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("Head") then return false end
    local origin = myChar.Head.Position
    local direction = targetPart.Position - origin
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {myChar, targetPart.Parent}
    local result = workspace:Raycast(origin, direction, raycastParams)
    return result == nil
end

local function getClosestEnemy()
    local target = nil
    local shortestDistance = math.huge
    for _, char in ipairs(getTargetCharacters()) do
        if isEnemy(char) then
            local targetPart = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
            if targetPart and isVisible(targetPart, getgenv().TWKS_Config.AimbotWallCheck) then
                local success, screenPoint, onScreen = pcall(function()
                    return CurrentCamera:WorldToViewportPoint(targetPart.Position)
                end)
                if success and onScreen then
                    local mouse = LocalPlayer:GetMouse()
                    local dist = (Vector2.new(mouse.X, mouse.Y) - Vector2.new(screenPoint.X, screenPoint.Y)).Magnitude
                    if dist < shortestDistance then
                        shortestDistance = dist
                        target = targetPart
                    end
                end
            end
        end
    end
    return target
end

-- РЕНДЕР И ЛОГИКА
local espDrawings = {}
local function clearESP()
    for _, drawings in pairs(espDrawings) do
        pcall(function()
            if drawings.Box then drawings.Box:Remove() end
            if drawings.Name then drawings.Name:Remove() end
        end)
    end
    espDrawings = {}
end

local cflyConnection = nil
local originalGravity = workspace.Gravity
local storedCFlyCFrame = nil
local originalWalkSpeed = 16

RunService.RenderStepped:Connect(function(dt)
    local cfg = getgenv().TWKS_Config
    
    CenterClickBtn.Visible = cfg.CenterClickEnabled
    for _, opt in ipairs(floatOptions) do
        local btn = activeFloatingButtons[opt]
        if btn then
            local show = cfg["FloatingButtons_" .. opt]
            btn.Visible = show
            if show then
                if cfg[opt] then
                    btn.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
                else
                    btn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
                end
            end
        end
    end
    
    -- Fly
    if cfg.Fly then
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local root = char.HumanoidRootPart
                workspace.Gravity = 0
                local moveDir = Vector3.new(0,0,0)
                if isFlyingUp then moveDir = moveDir + Vector3.new(0,1,0) end
                if isFlyingDown then moveDir = moveDir - Vector3.new(0,1,0) end
                root.AssemblyLinearVelocity = moveDir * cfg.FlySpeed
                FlyUpBtn.Visible = true
                FlyDownBtn.Visible = true
            end
        end)
    else
        FlyUpBtn.Visible = false
        FlyDownBtn.Visible = false
    end
    
    -- C-Fly
    if cfg.CFly and not cflyConnection then
        workspace.Gravity = 0
        storedCFlyCFrame = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character.HumanoidRootPart.CFrame
        cflyConnection = RunService.RenderStepped:Connect(function(delta)
            if not cfg.CFly then return end
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local root = char.HumanoidRootPart
                local moveDir = Vector3.new(0,0,0)
                if isFlyingUp then moveDir = moveDir + Vector3.new(0,1,0) end
                if isFlyingDown then moveDir = moveDir - Vector3.new(0,1,0) end
                if moveDir.Magnitude > 0 then
                    storedCFlyCFrame = storedCFlyCFrame + (moveDir.Unit * cfg.CFlySpeed * delta)
                end
                root.CFrame = storedCFlyCFrame
                root.AssemblyLinearVelocity = Vector3.new(0,0,0)
            end
        end)
    elseif not cfg.CFly and cflyConnection then
        cflyConnection:Disconnect()
        cflyConnection = nil
        workspace.Gravity = originalGravity
    end
    
    -- SpeedHack
    pcall(function()
        local char = LocalPlayer.Character
        if char and char:FindFirstChildOfClass("Humanoid") then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if originalWalkSpeed == 16 then originalWalkSpeed = hum.WalkSpeed end
            if cfg.SpeedHack and not cfg.Bunnyhop then
                hum.WalkSpeed = cfg.SpeedValue
            elseif not cfg.Bunnyhop then
                hum.WalkSpeed = originalWalkSpeed
            end
        end
    end)
    
    -- JumpHack
    pcall(function()
        local char = LocalPlayer.Character
        if char and char:FindFirstChildOfClass("Humanoid") then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if cfg.JumpHack then
                hum.JumpPower = cfg.JumpPower
                hum.UseJumpPower = true
            end
        end
    end)
    
    -- BunnyHop
    if cfg.Bunnyhop then
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChildOfClass("Humanoid") then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum.MoveDirection.Magnitude > 0 then
                    if hum.FloorMaterial ~= Enum.Material.Air then
                        hum:ChangeState(Enum.HumanoidStateType.Jumping)
                        hum.WalkSpeed = math.min(hum.WalkSpeed * cfg.BhopSpeedGain, 120)
                    end
                else
                    hum.WalkSpeed = originalWalkSpeed
                end
            end
        end)
    end
    
    -- SpinBot
    if cfg.SpinBot then
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local root = char.HumanoidRootPart
                local rot = tick() * cfg.SpinSpeed
                root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(rot % 360), 0)
            end
        end)
    end
    
    -- Aimbot
    local targetPart = getClosestEnemy()
    if cfg.Aimbot and targetPart then
        pcall(function()
            local currentPos = CurrentCamera.CFrame.Position
            local targetCFrame = CFrame.lookAt(currentPos, targetPart.Position)
            local sens = math.clamp(cfg.Sensitivity / 10, 0.05, 1)
            CurrentCamera.CFrame = CurrentCamera.CFrame:Lerp(targetCFrame, sens)
        end)
    end
    
    -- Triggerbot (с задержкой в миллисекундах)
    if cfg.Triggerbot then
        for _, char in ipairs(getTargetCharacters()) do
            if isEnemy(char) then
                local trigPart = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
                if trigPart and isVisible(trigPart, cfg.TriggerWallCheck) then
                    local success, screenPoint, onScreen = pcall(function()
                        return CurrentCamera:WorldToViewportPoint(trigPart.Position)
                    end)
                    if success and onScreen then
                        local viewport = CurrentCamera.ViewportSize
                        local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
                        if (center - Vector2.new(screenPoint.X, screenPoint.Y)).Magnitude <= 35 then
                            if cfg.TriggerDelay > 0 then
                                task.wait(cfg.TriggerDelay / 1000)
                            end
                            performClick()
                            break
                        end
                    end
                end
            end
        end
    end
    
    -- Chams (динамические цвета)
    local targets = getTargetCharacters()
    for _, char in ipairs(targets) do
        pcall(function()
            if cfg.Chams then
                local highlight = char:FindFirstChild("TWKS_Chams")
                if not highlight then
                    highlight = Instance.new("Highlight")
                    highlight.Name = "TWKS_Chams"
                    highlight.Adornee = char
                    highlight.FillTransparency = cfg.ChamsTransparency
                    highlight.Parent = char
                end
                local part = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildOfClass("BasePart")
                if part and isVisible(part, true) then
                    highlight.FillColor = hexToColor3(cfg.ChamsVisibleHex)
                    highlight.OutlineColor = hexToColor3(cfg.ChamsVisibleHex)
                else
                    highlight.FillColor = hexToColor3(cfg.ChamsHiddenHex)
                    highlight.OutlineColor = hexToColor3(cfg.ChamsHiddenHex)
                end
            else
                local highlight = char:FindFirstChild("TWKS_Chams")
                if highlight then highlight:Destroy() end
            end
        end)
    end
    
    -- ESP (динамические цвета)
    if not cfg.ESP_Enabled then
        clearESP()
    else
        for _, char in ipairs(targets) do
            pcall(function()
                local primaryPart = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildOfClass("BasePart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if primaryPart and humanoid and humanoid.Health > 0 then
                    if not espDrawings[char] then
                        espDrawings[char] = {
                            Box = Drawing.new("Square"),
                            Name = Drawing.new("Text")
                        }
                        espDrawings[char].Box.Thickness = 1
                        espDrawings[char].Box.Filled = false
                        espDrawings[char].Name.Size = 14
                        espDrawings[char].Name.Center = true
                        espDrawings[char].Name.Outline = true
                    end
                    
                    local d = espDrawings[char]
                    local isVis = isVisible(primaryPart, true)
                    local col = isVis and hexToColor3(cfg.ESP_VisibleHex) or hexToColor3(cfg.ESP_HiddenHex)
                    d.Box.Color = col
                    d.Name.Color = col
                    
                    local pos, onScreen = CurrentCamera:WorldToViewportPoint(primaryPart.Position)
                    if onScreen and pos.Z > 0 then
                        if cfg.Box2D then
                            d.Box.Visible = true
                            d.Box.Size = Vector2.new(35, 55)
                            d.Box.Position = Vector2.new(pos.X - 17.5, pos.Y - 27.5)
                        else
                            d.Box.Visible = false
                        end
                        if cfg.Names then
                            d.Name.Visible = true
                            d.Name.Text = char.Name
                            d.Name.Position = Vector2.new(pos.X, pos.Y - 40)
                        else
                            d.Name.Visible = false
                        end
                    else
                        d.Box.Visible = false
                        d.Name.Visible = false
                    end
                else
                    if espDrawings[char] then
                        if espDrawings[char].Box then espDrawings[char].Box:Remove() end
                        if espDrawings[char].Name then espDrawings[char].Name:Remove() end
                        espDrawings[char] = nil
                    end
                end
            end)
        end
    end
end)
